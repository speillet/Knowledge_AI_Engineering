{{- /* Tab and home-screen icons, replacing the Blowfish defaults of the same names in static/ (rendered from the SVG by `make icons`). The SVG URL carries the fingerprint of its content and the other icons a version derived from every icon file, so browsers fetch the new icons instead of the cached ones whenever one changes. The ICO comes first with its size, so Chrome still prefers the SVG. */ -}} {{- $svg := resources.Get "icons/favicon.svg" | fingerprint "sha256" -}} {{- $all := $svg.Content -}} {{- range slice "favicon.ico" "apple-touch-icon.png" "android-chrome-192x192.png" "android-chrome-512x512.png" "site.webmanifest" -}} {{- $all = print $all (readFile (print "static/" .)) -}} {{- end -}} {{- $v := substr (md5 $all) 0 10 -}}
